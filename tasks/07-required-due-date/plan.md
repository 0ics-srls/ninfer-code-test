**Approved design (user, 2026-09-06 — unchanged):**
1. Every todo HAS a due date, enforced by the backend: entity/DTO `required DateOnly DueDate`, column NOT NULL, POST/PUT without `dueDate` (or `null`) → 400. Frontend is only a view of that rule.
2. Week page = exactly 7 day columns Mon..Sun of a SELECTED week. No "Undated" column. Navigation: ‹ Prev / Today / Next › buttons + a week-range label ("31 Aug – 6 Sep 2026") + a date picker to jump to any week. Endpoint `GET /api/Todos/week?date=yyyy-MM-dd` (server normalizes to that week's Monday; no param = current UTC week). Frontend ALWAYS sends its local `selectedDate` (no UTC/local drift — this also fixes 06's defect "week computed in UTC on the server vs local on the client").
3. Create dialog: due date DEFAULTS TO TODAY (no clear button; Save enabled as soon as the title is valid). Each week column gets a "+" that opens the dialog with that column's date preselected.
4. Table gets a sortable "Due Date" column. Nav active link becomes an obvious filled pill + `aria-current="page"`.

function toIso(d: Date): string { return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`; }
export function addDaysIso(iso: string, days: number): string { const d = parseIsoDate(iso); d.setDate(d.getDate() + days); return toIso(d); }
export function mondayOfIso(iso: string): string { const d = parseIsoDate(iso); return addDaysIso(iso, -((d.getDay() + 6) % 7)); }
// Fixed table on purpose: en-GB ICU (Node 24 + Chromium) renders September as "Sept" via toLocaleDateString.
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
export function formatWeekRange(mondayIso: string): string {
  const mon = parseIsoDate(mondayIso), sun = parseIsoDate(addDaysIso(mondayIso, 6));
  const dm = (x: Date): string => `${x.getDate()} ${MONTHS[x.getMonth()]}`;

    ? `${dm(mon)} – ${dm(sun)} ${sun.getFullYear()}`
    : `${dm(mon)} ${mon.getFullYear()} – ${dm(sun)} ${sun.getFullYear()}`;
}
